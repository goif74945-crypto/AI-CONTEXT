cmake_minimum_required(VERSION 3.20)
project(mmcrc20 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(mmcrc20 src/court.cpp)
target_include_directories(mmcrc20 PUBLIC include)
target_compile_options(mmcrc20 PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(mmcrc20_tests tests/test_main.cpp)
target_link_libraries(mmcrc20_tests PRIVATE mmcrc20)
target_compile_options(mmcrc20_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(mmcrc20_integration examples/integration_main.cpp)
target_link_libraries(mmcrc20_integration PRIVATE mmcrc20)
target_compile_options(mmcrc20_integration PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME mmcrc20_tests COMMAND mmcrc20_tests)
add_test(NAME mmcrc20_integration COMMAND mmcrc20_integration)
